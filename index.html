  function supabaseHeaders(extra) {
    return Object.assign(
      {
        apikey: CONFIG.SUPABASE_KEY,
        Authorization:
          "Bearer " + CONFIG.SUPABASE_KEY,
        "Content-Type":
          "application/json"

  async function supabaseRequest(path, options) {
    const response = await fetch(
      CONFIG.SUPABASE_URL +
        "/rest/v1/" +
        CONFIG.TABLE +
        path,
      Object.assign(
        {
          headers:
            supabaseHeaders()
        },
        options || {}
      )
    );

    const text =
      await response.text();

    if (!response.ok) {
      throw new Error(
        "Supabase error " +
          response.status +
          ": " +
          (text || response.statusText)
      );
    }

    if (!text) {
      return null;
    }

    try {
      return JSON.parse(text);
    } catch (_) {
      return text;
    }
  }

      getSugarCube();

    /*
     * Put the backup into the exact storage

    sessionStorage.setItem(
      CONFIG.STORAGE_KEY,

          makeButton(
            "Restore"
          );


        const deleteButton =
          makeButton(
            "Delete"
          );


        restoreButton.onclick =
          async () => {

            restoreButton.disabled =
              true;

            try {

              await restoreBackup(
                backup
              );

            } finally {

              restoreButton.disabled =
                false;
            }
          };


        deleteButton.onclick =
          async () => {

            const confirmed =
              confirm(
                'Delete "' +
                  (
                    backup.slot_name ||
                    "Unnamed Save"
                  ) +
                  '"?'
              );


            if (!confirmed) {
              return;
            }


            deleteButton.disabled =
              true;


            try {

              await deleteBackup(
                backup.id
              );

            } catch (error) {

              alert(
                "Delete failed:\n\n" +
                  (
                    error.message ||
                    String(error)
                  )
              );

            } finally {

              deleteButton.disabled =
                false;
            }
          };


        row.appendChild(
          title
        );

        row.appendChild(
          date
        );

        row.appendChild(
          info
        );

        row.appendChild(
          restoreButton
        );

        row.appendChild(
          deleteButton
        );


        list.appendChild(
          row
        );
      }


      setStatus(
        rows.length +
          " backup" +
          (
            rows.length === 1
              ? ""
              : "s"
          ) +
          " found."
      );


    } catch (error) {

      console.error(
        error
      );


      setStatus(
        "Error: " +
          (
            error.message ||
            String(error)
          )
      );
    }
  }


  /* ============================================================
   * CLOSE
   * ============================================================ */

  panel.querySelector(
    "#ssm-close"
  ).onclick =
    () => {

      panel.style.display =
        "none";
    };


  /* ============================================================
   * REFRESH BUTTON
   * ============================================================ */

  panel.querySelector(
    "#ssm-refresh"
  ).onclick =
    () => {

      refreshList();
    };


  /* ============================================================
   * SAVE BUTTON
   * ============================================================ */

  panel.querySelector(
    "#ssm-save"
  ).onclick =
    async () => {

      const button =
        panel.querySelector(
          "#ssm-save"
        );


      button.disabled =
        true;


      try {

        await save();

      } catch (error) {

        console.error(
          error
        );


        alert(
          "Save failed:\n\n" +
            (
              error.message ||
              String(error)
            )
        );

      } finally {

        button.disabled =
          false;
      }
    };


  /* ============================================================
   * HOTKEY
   * ============================================================ */

  document.addEventListener(
    "keydown",
    event => {

      if (
        event.key !==
          CONFIG.HOTKEY
      ) {
        return;
      }


      if (
        event.ctrlKey ||
        event.altKey ||
        event.metaKey ||
        event.shiftKey
      ) {
        return;
      }


      const target =
        event.target;


      if (
        target &&
        (
          target.tagName ===
            "INPUT" ||
          target.tagName ===
            "TEXTAREA" ||
          target.isContentEditable
        )
      ) {
        return;
      }


      panel.style.display =
        panel.style.display ===
          "none"
            ? "block"
            : "none";
    }
  );


  /* ============================================================
   * CONSOLE API
   * ============================================================ */

  window.gameSaveManager = {

    detect,

    save,

    backups,

    all:
      allBackups,

    restore:
      restoreBackup,

    restoreById,

    restoreLive,

    refresh:
      refreshList,

    config:
      CONFIG
  };


  /* ============================================================
   * START
   * ============================================================ */

  console.log(
    "%cSociety Save Manager loaded.",
    "font-weight:bold;font-size:14px"
  );


  console.log(
    "Hotkey:",
    CONFIG.HOTKEY
  );


  console.log(
    "Console API:"
  );


  console.log(
    "gameSaveManager.detect()"
  );


  console.log(
    "gameSaveManager.save()"
  );


  console.log(
    "gameSaveManager.backups()"
  );


  console.log(
    "gameSaveManager.restore()"
  );


  console.log(
    "gameSaveManager.restoreById(id)"
  );


  console.log(
    "gameSaveManager.all()"
  );


  refreshList();

})();
